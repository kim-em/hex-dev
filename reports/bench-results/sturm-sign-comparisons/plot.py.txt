"""Plot retained observations; quadratic guide is anchored, never fitted."""
from pathlib import Path
import argparse, json, statistics
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from matplotlib.ticker import NullLocator

p=argparse.ArgumentParser(); p.add_argument('capture', type=Path); args=p.parse_args(); out=args.capture
rows=[json.loads(line) for line in (out/'paired.jsonl').read_text().splitlines()]
summary=json.loads((out/'summary.json').read_text())
ladder=json.loads((out/'after-ladder.json').read_text())['results'][0]
fig, axes=plt.subplots(1,2,figsize=(11,4.4),layout='constrained')
colors={'before':'#b45309','after':'#0369a1'}
for arm in colors:
 samples=[r for r in rows if r['arm']==arm]
 axes[0].scatter([r['param'] for r in samples],[r['measurement']['per_call_nanos']/1e6 for r in samples],color=colors[arm],alpha=.35,s=22)
 axes[0].plot([r['param'] for r in summary],[r['median_nanos'][arm]/1e6 for r in summary],label=arm+' (paired medians)',marker='o',color=colors[arm])
params=[s['param'] for s in summary]
anchor=summary[0]['median_nanos']['after']/1e6
axes[0].plot(params,[anchor*(n/params[0])**2 for n in params],':',color='#64748b',label='quadratic guide at after n=128')
axes[0].set(xscale='log',yscale='log',xlabel='Chebyshev head degree n',ylabel='Sign traversal + result hash (ms)',title='Bench sign traversal after compiler change')
for point in ladder['points']:
 axes[1].scatter(point['param'],point['per_call_nanos']/point['param']**2,color=colors['after'],alpha=.45,s=26)
medians=[s['median_per_call_nanos']/s['param']**2 for s in ladder['trial_summaries']]
repeat=json.loads((out.parent/'rerun/results.json').read_text())['results'][0]
for point in repeat['points']:
 axes[1].scatter(point['param'],point['per_call_nanos']/point['param']**2,color='#15803d',alpha=.45,s=26)
axes[1].plot(params,[s['median_per_call_nanos']/s['param']**2 for s in repeat['trial_summaries']], 'o-',color='#15803d',label='one unchanged repeat')
axes[1].plot(params,medians,'o-',color=colors['after'],label='registered after ladder')
axes[1].set(xscale='log',xlabel='Chebyshev head degree n',ylabel='Nanoseconds / n²',title='Unchanged quadratic model: '+ladder['verdict'])
for ax in axes:
 ax.set_xticks(params,labels=[str(n) for n in params]); ax.xaxis.set_minor_locator(NullLocator()); ax.grid(alpha=.2); ax.legend(fontsize=8)
fig.savefig(out/'sign-scaling.svg'); fig.savefig(out/'sign-scaling.png',dpi=160)

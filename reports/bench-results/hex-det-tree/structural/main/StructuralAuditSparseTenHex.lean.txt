import HexPolyDetMathlib.ProofProbe.StructuralSparseTenHex
#print axioms HexPolyDetMathlib.ProofProbe.StructuralSparseTenHex.result
run_meta do
  let root := ``HexPolyDetMathlib.ProofProbe.StructuralSparseTenHex.result
  let mut pending := [root]
  let mut found := false
  while let name :: rest := pending do
    pending := rest
    let some value := (← Lean.getConstInfo name).value? (allowOpaque := true)
      | throwError "missing determinant proof"
    for used in value.getUsedConstants do
      if used == ``HexPolyDetMathlib.Structural.cofactor then found := true
      if root.isPrefixOf used then pending := used :: pending
  unless found do throwError "intended structural route was not used"

#!/usr/bin/env python3
"""Generate a complete proof using the fixed quadratic witness and cached arithmetic."""
import json
from pathlib import Path
import sys

from arithmetic import polynomial


def poly(p):
    return polynomial(p).replace(': Int', ': S')


def row(ps):
    return '[' + ', '.join(map(poly, ps)) + ']'


def dot(a, b):
    tail = '0'
    for x, y in reversed(list(zip(a, b))):
        tail = f'({poly(x)} * {poly(y)} + {tail})'
    return tail


def generate(obligations):
    ts = [obligations[i][0] for i in (0, 1, 3, 6)]
    a = list(map(list, zip(*(o[1][0] for o in obligations[-4:]))))
    d = obligations[-1][1][1]
    rows = '[' + ', '.join(map(row, a)) + ']'
    transform = '[' + ', '.join(map(row, ts)) + ']'
    identities = [f'{dot(l, r)} = {poly(v)}' for l, (r, v) in obligations]
    pivots = [poly(ts[i][-1]) for i in (1, 2, 3)]
    code = '''/-
Copyright (c) 2026 Lean FRO, LLC. All rights reserved.
Released under Apache 2.0 license as described in the file LICENSE.
Authors: Kim Morrison
-/
-- Generated by experiments/Determinant/full.py from the fixed witness.
import Determinant.Arithmetic
import HexPolyDetMathlib.Tree
open Determinant HexMatrixMathlib.DetPoly
set_option maxHeartbeats 500000
set_option maxRecDepth 10000
set_option profiler false
namespace Determinant.Full
variable {S : Type} [CommRing S]
'''
    code += 'theorem products (x0 x1 : S) :\n  ' + ' ∧\n  '.join(f'({e})' for e in identities) + ''' ∧ True := by
  cached_ring
'''
    code += 'theorem witness [IsDomain S] (x0 x1 : S)\n'
    code += '\n'.join(f'    (h{i} : {p} ≠ 0)' for i, p in enumerate(pivots))
    code += f' :\n    ((Decode.identity S).matrix 4 {rows}).det = {poly(d)} := by\n'
    code += '  rcases products x0 x1 with ⟨p0, p1, p2, p3, p4, p5, p6, p7, p8, p9, _⟩\n'
    code += f'  apply (Decode.identity S).identities_sound 4 {rows} (.triangular [] {transform} {poly(d)})\n'
    code += '''  refine ⟨rfl, rfl, rfl, rfl, rfl,
    rfl, rfl, one_ne_zero, rfl, ?_,
    rfl, rfl, h0, p0.symm, ?_,
    rfl, rfl, h1, p2.symm, ?_,
    rfl, rfl, h2, p5.symm, ?_, p9.symm⟩
  · intro c hc
    cases hc
  · intro c hc
    simp only [List.mem_cons, List.not_mem_nil, or_false] at hc
    rcases hc with rfl | hc
    · exact p1
  · intro c hc
    simp only [List.mem_cons, List.not_mem_nil, or_false] at hc
    rcases hc with rfl | rfl | hc
    · exact p4
    · exact p3
  · intro c hc
    simp only [List.mem_cons, List.not_mem_nil, or_false] at hc
    rcases hc with rfl | rfl | rfl | hc
    · exact p8
    · exact p7
    · exact p6
'''
    code += '''abbrev P := MvPolynomial (Fin 2) Int
'''
    literal = '!![' + '; '.join(', '.join(map(poly, r)) for r in a) + ']'
    goal = f'Matrix.det ({literal}) = {poly(d)}'
    model_goal = goal.replace('x0', '(MvPolynomial.X (0 : Fin 2) : P)').replace('x1', '(MvPolynomial.X (1 : Fin 2) : P)').replace(': S', ': P')
    code += f'theorem model : {model_goal} := by\n'
    code += '''  apply witness (MvPolynomial.X 0) (MvPolynomial.X 1)
  all_goals
    intro h
    have he := congrArg (MvPolynomial.eval₂Hom (RingHom.id Int) (fun _ : Fin 2 => 0)) h
    simp only [map_add, map_mul, map_neg, map_pow, map_ofNat, map_one, map_zero,
      MvPolynomial.eval₂Hom_X', RingHom.id_apply] at he
    norm_num at he
'''
    code += f'theorem result (x0 x1 : S) : {goal} := by\n'
    model_matrix = literal.replace('x0', '(MvPolynomial.X (0 : Fin 2) : P)').replace('x1', '(MvPolynomial.X (1 : Fin 2) : P)').replace(': S', ': P')
    code += "  let φ := MvPolynomial.eval₂Hom (Int.castRingHom S) ![x0, x1]\n"
    code += "  have h := congrArg φ model\n  rw [RingHom.map_det] at h\n"
    code += f"  have hm : ({model_matrix}).map φ = {literal} := by\n"
    code += "    ext i j; fin_cases i <;> fin_cases j <;> simp [Matrix.map_apply, φ]\n"
    code += f"  change (({model_matrix}).map φ).det = _ at h\n"
    code += "  rw [hm] at h\n"
    code += "  simpa only [φ, map_add, map_mul, map_neg, map_pow, map_ofNat, map_one, map_zero,\n    MvPolynomial.eval₂Hom_X', Matrix.cons_val_zero, Matrix.cons_val_one] using h\n"
    code += "#print axioms result\nend Determinant.Full\n"
    return code


if __name__ == '__main__':
    obligations = json.loads(Path(sys.argv[1]).read_text())
    Path(sys.argv[2]).write_text(generate(obligations))

#!/usr/bin/env python3
"""Recompute the search inventory from retained observations, without measuring."""
import json
from pathlib import Path
import statistics
import sys
import re


def theorem_axioms(output):
    matches = re.findall(r"'Determinant\.(?:Adversarial|Full)\.result' depends on axioms: \[([^]]*)\]", output)
    assert len(matches) == 1, 'missing or ambiguous result theorem axiom audit'
    return [name.strip() for name in matches[0].split(',') if name.strip()]


def profile_components(output, module):
    """Lean groups several profiler lines under one source-location prefix."""
    prefix = f'info: experiments/Determinant/{module}.lean:'
    active, components = False, {}
    for line in output.splitlines():
        if line.startswith(('info: ', 'warning: ', 'error: ', 'ℹ ', '✔ ', '✖ ', 'Build completed')):
            active = line.startswith(prefix)
        if active:
            match = re.search(r'(type checking|share common exprs|fix level params|process pre-definitions) took ([0-9.e+]+)(ms|s)$', line)
            if match:
                name, value, unit = match.groups()
                components.setdefault(name, []).append(float(value) * (1000 if unit == 's' else 1))
    return components


def main(root):
    rows, elapsed, hashes, profiles = [], 0, {}, []
    for path in sorted(root.glob('*/case.json')):
        folder = path.parent
        case = json.loads(path.read_text())
        status = json.loads((folder / 'status.json').read_text())
        elapsed += status['elapsed_seconds']
        result = folder / 'results.json'
        data = json.loads(result.read_text()) if result.exists() else {'samples': []}
        samples = data['samples']
        for sample in samples:
            axioms = theorem_axioms(sample['result']['compiler_output'])
            assert set(axioms) <= {'propext', 'Classical.choice', 'Quot.sound'}
            assert set(axioms) == set(sample['result']['axioms'])
            if case.get('kernel_profile'):
                components = profile_components(sample['result']['compiler_output'], 'AdversarialSample' + sample['arm'])
                assert len(components.get('type checking', [])) == 1, components
                profiles.append(dict(case=folder.name, arm=sample['arm'], pair=sample['pair'],
                                     declaration_ms=sample['declaration_ms'], tactic_ms=sample['tactic_ms'], components_ms=components))
        stats = {}
        for arm in ['Mathlib', 'Shared']:
            xs = [s['declaration_ms'] / 1000 for s in samples if s['arm'] == arm]
            stats[arm] = dict(count=len(xs), median=statistics.median(xs), minimum=min(xs), maximum=max(xs)) if xs else None
        state = 'complete' if status['exit_status'] == 0 else 'failed'
        failure = folder / 'failure.json'
        if failure.exists():
            f = json.loads(failure.read_text())
            arm = 'Mathlib' if 'Mathlib' in f['command'][-1] else 'Shared'
            state = f'{arm} {f["state"]}'
            if 'Invalid field `setNat`' in f['compiler_output']:
                state = 'harness error (fixed profiler option)'
            if 'No goals to be solved' in f['compiler_output']:
                state = 'harness error (fixed comparator sequencing)'
        pairs = [{s['arm']: s['declaration_ms'] for s in samples if s['pair'] == pair} for pair in range(2)]
        losses = sum(p['Shared'] > p['Mathlib'] for p in pairs if len(p) == 2)
        sources = folder / 'sources.json'
        if sources.exists():
            for name, digest in json.loads(sources.read_text()).items():
                if name.endswith(('Deferred.lean', 'Arithmetic.lean')):
                    hashes.setdefault(name, set()).add(digest)
        rows.append(dict(name=folder.name, case=case, state=state, stats=stats,
                         paired_losses=losses, elapsed_seconds=status['elapsed_seconds']))
    assert all(len(values) == 1 for values in hashes.values()), 'prototype changed during search'
    summary = dict(elapsed_seconds=elapsed, cases=rows,
                   prototype_hashes={k: next(iter(v)) for k, v in hashes.items()})
    (root / 'summary.json').write_text(json.dumps(summary, indent=2) + '\n')
    (root / 'profiles.json').write_text(json.dumps(profiles, indent=2) + '\n')
    lines = ['# Adversarial proof search inventory', '',
             'Generated by `experiments/Determinant/adversarial_report.py` from every retained case.',
             'Times are complete declaration seconds: median [minimum, maximum].',
             'A single completed sample is labelled; failed builds are never successful timing samples.',
             'Timeouts are the 60-second fresh-module process ceiling, including loading/build overhead.', '',
             '| Case | State | Mathlib | Shared prototype | Slower prototype pairs |',
             '|---|---|---:|---:|---:|']
    for row in rows:
        def cell(arm):
            s = row['stats'][arm]
            if s is None:
                return '—'
            if s['count'] == 1:
                return f'{s["median"]:.3f} (one sample)'
            return f'{s["median"]:.3f} [{s["minimum"]:.3f}, {s["maximum"]:.3f}]'
        lines.append(f'| [{row["name"]}]({row["name"]}/case.json) | {row["state"]} | {cell("Mathlib")} | {cell("Shared")} | {row["paired_losses"]} |')
    lines += ['', f'Total retained measurement-batch wall time: {elapsed:.1f} seconds.',
              'All successful theorem dependencies are permitted, and both prototype source hashes are unchanged.', '']
    (root / 'inventory.md').write_text('\n'.join(lines))
    print(f'{len(rows)} cases; {elapsed:.1f} seconds; prototype hashes unchanged')


if __name__ == '__main__':
    main(Path(sys.argv[1]))

import HexManual
import HexSturmMathlib.Tests
import HexSturmMathlib.Tests.Replay.Accepted
import HexSturmMathlib.Tests.Replay.Rejected
import HexSturmMathlib.Tests.Replay.Baseline
import HexSturmMathlib.Tests.Replay.Semantics
import HexSturmMathlib.Tests.Replay.SemanticsBaseline
import HexSturm.Conformance
import Lean.Util.FoldConsts
import Lean.Util.CollectAxioms

open Lean Elab Command

set_option maxHeartbeats 0
set_option maxRecDepth 4096

run_cmd do
  let env ← getEnv
  let modules := ["HexSturm.Basic", "HexSturm.Reduced", "HexSturm.Transport", "HexSturm.DomainOperations", "HexSturmMathlib.Domain", "HexSturmMathlib.Soundness", "HexSturmMathlib.Reduced", "HexSturmMathlib.Compare", "HexSturmMathlib.Rational", "HexSturmMathlib.IntCast", "HexSturmMathlib.DenominatorClearing"]
  let constants := env.constants.toList
  let selected := constants.filter fun (name, _) =>
    match env.getModuleIdxFor? name with
    | none => false
    | some idx => modules.contains env.allImportedModuleNames[idx.toNat]!.toString
  let wanted : NameSet := .ofList (selected.map Prod.fst)
  let mut uses : Std.HashMap Name (Array Name) := {}
  for (name, info) in constants do
    for target in info.getUsedConstantsAsSet do
      if wanted.contains target && target != name then
        uses := uses.insert target ((uses.getD target #[]).push name)
  let mut rows : Array Json := #[]
  for (name, info) in selected do
    let axioms ← liftCoreM <| collectAxioms name
    let kind := match info with
      | .thmInfo _ => "theorem"
      | .defnInfo _ => "definition"
      | .opaqueInfo _ => "opaque"
      | .axiomInfo _ => "axiom"
      | .inductInfo _ => "inductive"
      | .ctorInfo _ => "constructor"
      | .recInfo _ => "recursor"
      | .quotInfo _ => "quotient"
    let moduleName := env.allImportedModuleNames[(env.getModuleIdxFor? name).get!.toNat]!
    rows := rows.push <| Json.mkObj [
      ("name", toJson name.toString), ("module", toJson moduleName.toString),
      ("kind", toJson kind), ("users", toJson ((uses.getD name #[]).map Name.toString)),
      ("axioms", toJson (axioms.map Name.toString))]
  liftIO <| IO.FS.writeFile "reports/sturm-declaration-review/declaration-use.json"
    (Json.pretty (toJson rows) ++ "\n")
  logInfo m!"Recorded {rows.size} production constants from the actual imported environment."

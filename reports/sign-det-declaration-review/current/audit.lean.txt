/-- Resolve an existing declaration ancestor, preserving private internal names. -/
private def declarationBase (env : Environment) (name : Name) : Json × Json := Id.run do
  let mut raw := name.getPrefix
  let mut decoded := (privateToUserName name).getPrefix
  while !raw.isAnonymous || !decoded.isAnonymous do
    for candidate in [raw, decoded] do
      if let some idx := env.getModuleIdxFor? candidate then
        return (toJson candidate.toString,
          toJson env.allImportedModuleNames[idx.toNat]!.toString)
    raw := raw.getPrefix
    decoded := decoded.getPrefix
  return (Json.null, Json.null)

run_cmd do
  let env ← getEnv
  let output ← liftIO <| IO.getEnv "HEX_SIGNDET_AUDIT_OUTPUT"
  let some directory := output |
    throwError "Set HEX_SIGNDET_AUDIT_OUTPUT to an explicit capture directory"
  let modules := ["HexSignDetMathlib.Basis", "HexSignDetMathlib.CommonProduct", "HexSignDetMathlib.ComparisonProducer", "HexSignDetMathlib.CompletionProducer", "HexSignDetMathlib.Convert", "HexSignDetMathlib.DagSelectedSigns", "HexSignDetMathlib.Derivatives", "HexSignDetMathlib.Embedding", "HexSignDetMathlib.FiniteSolve", "HexSignDetMathlib.Foundation", "HexSignDetMathlib.Naturality", "HexSignDetMathlib.NodeBasis", "HexSignDetMathlib.NodeChecks", "HexSignDetMathlib.NodeProducer", "HexSignDetMathlib.NodeSolve", "HexSignDetMathlib.ParentSystem", "HexSignDetMathlib.QueryHandle", "HexSignDetMathlib.QueryReduction", "HexSignDetMathlib.RationalSolve", "HexSignDetMathlib.Reduction", "HexSignDetMathlib.ReductionProducer", "HexSignDetMathlib.Reencode", "HexSignDetMathlib.ReencodingProducer", "HexSignDetMathlib.ReencodingRefinement", "HexSignDetMathlib.RootList", "HexSignDetMathlib.RootModel", "HexSignDetMathlib.RootProducer", "HexSignDetMathlib.SelectedProducer", "HexSignDetMathlib.SelectedRoot", "HexSignDetMathlib.Solve", "HexSignDetMathlib.TableProducer", "HexSignDetMathlib.Tensor", "HexSignDetMathlib.Thom", "HexSignDetMathlib.ThomReencoding", "HexSignDetMathlib.ThomRoots", "HexSignDetMathlib.TreeChecks", "HexSignDetMathlib.TreeSolve"]
  let constants := env.constants.toList
  let selected := constants.filter fun (name, _) =>
    match env.getModuleIdxFor? name with
    | none => false
    | some idx => modules.contains env.allImportedModuleNames[idx.toNat]!.toString
  let mut fields : Array Json := #[]
  for (name, _) in selected do
    if (getStructureInfo? env name).isSome then
      for field in getStructureFields env name do
        let info := (getFieldInfo? env name field).get!
        fields := fields.push <| Json.mkObj [
          ("structure", toJson name.toString), ("field", toJson field.toString),
          ("projection", toJson info.projFn.toString)]
  liftIO <| IO.FS.writeFile (directory ++ "/structure-fields.json")
    (Json.pretty (toJson fields) ++ "\n")
  let wanted : NameSet := .ofList (selected.map Prod.fst)
  let mut uses : Std.HashMap Name (Array Name) := {}
  for (name, info) in constants do
    for target in info.getUsedConstantsAsSet do
      if wanted.contains target && target != name then
        uses := uses.insert target ((uses.getD target #[]).push name)
  let mut rows : Array Json := #[]
  for (name, info) in selected do
    let axioms ← liftCoreM <| collectAxioms name
    let kind := match info with
      | .thmInfo _ => "theorem"
      | .defnInfo _ => "definition"
      | .opaqueInfo _ => "opaque"
      | .axiomInfo _ => "axiom"
      | .inductInfo _ => "inductive"
      | .ctorInfo _ => "constructor"
      | .recInfo _ => "recursor"
      | .quotInfo _ => "quotient"
    let moduleName := env.allImportedModuleNames[(env.getModuleIdxFor? name).get!.toNat]!
    let (baseDeclaration, baseModule) := declarationBase env name
    let userNames := uses.getD name #[]
    let userModules := userNames.map fun user =>
      let origin := match env.getModuleIdxFor? user with
        | some idx => env.allImportedModuleNames[idx.toNat]!.toString
        | none => "current audit module"
      (user.toString, toJson origin)
    let userBases := userNames.map fun user =>
      let (base, origin) := declarationBase env user
      (user.toString, Json.mkObj [("declaration", base), ("module", origin)])
    rows := rows.push <| Json.mkObj [
      ("name", toJson name.toString), ("user_name", toJson (privateToUserName name).toString),
      ("module", toJson moduleName.toString),
      ("kind", toJson kind), ("users", toJson (userNames.map Name.toString)),
      ("base_declaration", baseDeclaration), ("base_module", baseModule),
      ("user_modules", Json.mkObj userModules.toList),
      ("user_bases", Json.mkObj userBases.toList),
      ("axioms", toJson (axioms.map Name.toString))]
  liftIO <| IO.FS.writeFile (directory ++ "/declaration-use.json")
    (Json.pretty (toJson rows) ++ "\n")
  liftIO <| IO.FS.writeFile (directory ++ "/audit-imports.json")
    (Json.pretty (toJson (env.allImportedModuleNames.map Name.toString)) ++ "\n")
  logInfo m!"Recorded {rows.size} constants attributed by Lean to the 37 companion production modules."

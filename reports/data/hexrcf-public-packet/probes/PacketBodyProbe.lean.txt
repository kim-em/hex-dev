private partial def shape (e : Expr) (depth : Nat) : String :=
  if depth = 0 then s!"{e.getAppFn.constName?}/{e.getAppNumArgs}"
  else match e with
  | .lam name type body _ => s!"lam({name}: {shape type (depth-1)}), {shape body (depth-1)}"
  | .forallE name type body _ => s!"forall({name}: {shape type (depth-1)}), {shape body (depth-1)}"
  | .letE name type val body _ => s!"let {name}: {shape type (depth-1)} := {shape val (depth-1)}; {shape body (depth-1)}"
  | .mdata _ body => shape body (depth-1)
  | .proj name index body => s!"proj({name},{index},{shape body (depth-1)})"
  | .app .. => s!"app({e.getAppFn.constName?},[{String.intercalate ";" ((e.getAppArgs.extract 0 (min e.getAppNumArgs 8)).toList.map (fun a => shape a (depth-1)))}])"
  | _ => s!"{e}"

  IO.FS.writeFile "/home/kim/.codex/tasks/hex-10358/public-audit-345-summary.txt"
    s!"name={name}\ntype={shape info.type 5}\nvalue={shape value 5}\n"
  IO.FS.writeFile "/home/kim/.codex/tasks/hex-10358/public-audit-345-phase.txt" "shape written"

    IO.FS.writeFile "/home/kim/.codex/tasks/hex-10358/public-audit-345-value.txt" (← ppExpr value).pretty
  logInfo s!"body345 inspected: {name}"

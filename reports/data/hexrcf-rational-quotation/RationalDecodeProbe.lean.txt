private def number? (expression : Expr) : MetaM (Option Int) := do
  Core.checkSystem "rational literal number"
  let expression ← whnf expression
  unless expression.isAppOfArity ``Codec.Json.Value.number 1 do return none
  let argument := expression.appArg!
  if let some number ← getIntValue? argument then return some number
  let argument ← whnf argument
  if argument.isAppOfArity ``Int.ofNat 1 then
    return (← getNatValue? argument.appArg!).map Int.ofNat
  if argument.isAppOfArity ``Int.negSucc 1 then
    return (← getNatValue? argument.appArg!).map Int.negSucc
  return none

private def payload? (expression : Expr) : MetaM (Option (Int × Int)) := do
  let expression ← whnf expression
  unless expression.isAppOfArity ``Codec.Json.Value.array 1 do return none
  let mut rest := expression.appArg!
  let mut numbers : Array Int := #[]
  for _ in [:3] do
    Core.checkSystem "rational literal payload"
    let list ← whnf rest
    unless list.isAppOfArity ``Codec.Json.Values.cons 2 do return none
    let arguments := list.getAppArgs
    let some number ← number? arguments[0]! | return none
    numbers := numbers.push number
    rest := arguments[1]!
  unless (← whnf rest).isConstOf ``Codec.Json.Values.nil do return none
  unless numbers[0]! == 0 do return none
  return some (numbers[1]!,numbers[2]!)

def decoder (registry : Expr) : RootQuote.Decoder := fun expression => do
  if expression.hasMVar || registry.hasMVar then return none
  let .proj ``ValueCodec 1 _ := expression.getAppFn | return none
  unless expression.getAppNumArgs == 1 do return none
  let some (numerator,denominator) ← payload? expression.appArg! | return none
  unless denominator > 0 do return none
  let rational := mkRat numerator denominator.toNat
  unless rational.num == numerator && (rational.den : Int) == denominator do return none
  let value ← mkAppM ``mkRat #[mkIntLit numerator,mkNatLit denominator.toNat]
  let numeratorType ← mkEq (← mkAppM ``Rat.num #[value]) (mkIntLit numerator)
  let denominatorType ← mkEq (← mkAppM ``Int.ofNat #[← mkAppM ``Rat.den #[value]])
    (mkIntLit denominator)
  let proof ← mkAppM ``Hex.RealClosure.NumberField.decode_payload
    #[registry,value,mkIntLit numerator,mkIntLit denominator,
      ← mkDecideProof numeratorType,← mkDecideProof denominatorType]
  let type ← inferType proof
  if ← withNewMCtxDepth <| withTransparency .all <| isDefEq expression type.getAppArgs[1]! then
    return some proof
  return none

set_option maxHeartbeats 12000000
set_option maxRecDepth 8192

private def scalar (n d : Int) (tag : Int := 0) : Expr :=
  let numbers := [tag,n,d].foldr (fun i rest =>
    mkApp2 (mkConst ``Codec.Json.Values.cons)
      (mkApp (mkConst ``Codec.Json.Value.number) (mkIntLit i)) rest)
    (mkConst ``Codec.Json.Values.nil)
  mkApp (mkConst ``Codec.Json.Value.array) numbers

run_meta do
  withExporting (isExporting := false) do
    let registry := mkConst ``Hex.RCF.RootQuoteTests.registry
    let context ← mkAppM ``Hex.RealClosure.NumberField.base #[registry]
    let codec ← mkAppM ``Tower.Context.codec #[context]
    for (n,d) in ([(0,1),(7,3),(-7,3),(17,13),(123456789,100000000)] : List (Int × Int)) do
      let reader := mkApp (.proj ``ValueCodec 1 codec) (scalar n d)
      let some proof ← decoder registry reader | throwError "canonical fraction declined: {n}/{d}"
      let type ← inferType proof
      unless proof.getUsedConstants.contains ``Hex.RealClosure.NumberField.decode_payload do
        throwError "fraction decoder did not use the owner equation"
      let statement ← mkEq reader type.getAppArgs[2]!
      let _ ← Hex.RCF.checkProof `canonicalFraction statement (mkExpectedPropHint proof statement)
      let .ok result ← RootQuote.quote reader (decoder registry)
        | throwError "canonical fraction quotation failed: {n}/{d}"
      let _ ← Hex.RCF.checkProof `quotedFraction (← inferType result.proof) result.proof
    for (n,d,tag) in ([(0,0,0),(2,4,0),(7,-3,0),(7,3,1)] : List (Int × Int × Int)) do
      let reader := mkApp (.proj ``ValueCodec 1 codec) (scalar n d tag)
      unless (← decoder registry reader).isNone do throwError "malformed/noncanonical fraction accepted"
    let wrong := mkApp (.proj ``ValueCodec 1 (mkConst ``ValueCodec.rat)) (scalar 7 3)
    unless (← decoder registry wrong).isNone do throwError "different coefficient codec accepted"
    logInfo "generic decoder accepts checked fractions and declines noncanonical payloads or a different codec"

run_meta do
  withExporting (isExporting := false) do
    let .ok result ← RootQuote.quote (mkConst ``Hex.RCF.RootQuoteTests.read)
        (decoder (mkConst ``Hex.RCF.RootQuoteTests.registry))
      | throwError "generic rational decoder did not quote the frozen cubic"
    let _ ← Hex.RCF.checkProof `genericCubic (← inferType result.proof) result.proof
    logInfo m!"generic rational decoder quoted frozen cubic, proofNodes={result.usage.proofNodes}"

end RationalDecodeProbe

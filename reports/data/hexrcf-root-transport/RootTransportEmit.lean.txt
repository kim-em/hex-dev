import HexRCF.RootTransportData
import HexRCF.SelectedRoot.Data
import HexSignDet.DagEncode
open Hex Hex.RealClosure Hex.RealClosure.Tower Hex.SignDet
open Hex.RCF.SelectedRootTests Hex.RCF.RealCoefficients
namespace RootTransportEmit
private def pair (variant : Nat) : Codec.Json :=
  let c := RootTransportData.conversion
  let source := Data.base.mapDescriptor c.context c.value Data.lower
  let raw := if variant = 1 then
      {source with lower := .finite (c.value (-2)), upper := .finite (c.value (-1))}
    else if variant = 2 then {source with indices := [1], signs := [1]}
    else source
  match Descriptor.validate c.context.sign c.context.signature raw with
  | none => .null
  | some root =>
    letI : Hashable c.context.Value := ⟨fun a => hash (c.context.codec.encode a)⟩
    letI : Hashable Signature := ⟨fun a => hash (Signature.codec.encode a)⟩
    .arr #[Algebraic.SignRequests.binding c.context.codec Signature.codec root.raw,
      Codec.graph c.context.codec Signature.codec (Dag.encode root.evidence)]
private def packet : Codec.Json :=
  .arr ((#[pair 0,pair 1,pair 2]).flatMap fun p =>
    match p with | .array fields => fields.toArray | _ => #[])
#eval IO.println ("ROOT_TRANSPORT_PACKET:" ++ String.fromUTF8! packet.writeBytes)
end RootTransportEmit

theorem context_roundtrip (parent context : Tower.Signature) :
    (Tower.contextCodec parent).decode ((Tower.contextCodec parent).encode context) = .ok context :=
  Tower.contextCodec_lawful parent context

theorem value_roundtrip (value : base.Value) :
    base.codec.decode (base.codec.encode value) = .ok value := base.codec_lawful value

open Lean Meta in
run_meta do
  let program ← mkAppM ``Except.toOption #[mkConst ``read]
  let original := program
  let initial ← withTransparency .all (whnf original)
  let mut rules : SimpTheorems := {}
  rules ← rules.addConst ``value_roundtrip
  rules ← rules.addConst ``context_roundtrip
  let context ← Simp.mkContext (simpTheorems := #[← getSimpTheorems, rules])
    (config := {decide := false, maxSteps := 100000})
    (congrTheorems := ← getSimpCongrTheorems)
  let (simplified,_) ← withTransparency .all (Meta.simp initial context)
  let rewrite ← simplified.getProof' initial
  let equality ← mkEq original simplified.expr
  let _ ← Hex.RCF.checkProof `atomicRoundtripReduction equality (mkExpectedPropHint rewrite equality)
  let (record,raw) ← Hex.RealClosure.Algebraic.KernelReplay.checkedRecordProof simplified.expr
  let accepted ← mkEq simplified.expr (← mkAppM ``Option.some #[record])
  let _ ← Hex.RCF.checkProof `atomicRoundtripAcceptance accepted (mkExpectedPropHint raw accepted)
  logInfo "public-lawful codec roundtrips and the existing literal reader accepted an independent cubic descriptor"
end AtomicRoundtripProbe

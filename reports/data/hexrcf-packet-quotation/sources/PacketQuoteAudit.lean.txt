import PacketQuoteSemantics
import HexRCF.ProofEvidence
import KernelReplay.AxiomClosure
import HexRCF.SelectedRoot.KernelCheck
open Lean Meta Hex Hex.RealClosure Hex.SignDet Hex.RCF.RealCoefficients
namespace Hex.RCF.PacketQuoteTests
set_option maxHeartbeats 12000000
set_option maxRecDepth 32768
run_meta do
  let roots := #[``literal_read,``input_read,``input_size,``reduction_eq,``table_read,``roots_count,``witness]
  let axioms ← Algebraic.KernelReplay.AxiomClosure.collect roots
  unless axioms == (#[``propext,``Classical.choice,``Quot.sound].qsort Name.lt) do
    throwError "unexpected complete packet quotation axiom closure: {axioms}"
  let dependencies ← Hex.RCF.ProofEvidence.projectDependencies roots
  for required in #[``RootTable.read,``RootTable.extract_eq,``Dag.replay_eq,
      ``Codec.readPoly_refines,``Algebraic.Element.signCodec_sound,
      ``CellReplayTests.Algebraic.Records.accepted0] do
    unless dependencies.contains required do throwError "packet quote omitted evidence: {required}"
  for forbidden in #[``CellReplayTests.Algebraic.Inputs.polynomialAt,
      ``ScalarPacket.build,``ScalarPacket.buildLiteral] do
    if dependencies.contains forbidden then throwError "packet quote retained source computation: {forbidden}"
  let modules := (← getEnv).header.moduleNames
  let mut theorems : Array Name := #[]
  let mut definitions : Array Name := #[]
  let mut reductions : Nat := 0
  for name in dependencies.toArray do
    if let some index := (← getEnv).getModuleIdxFor? name then
      if #[`PacketQuoteProbe,`PacketQuoteInput,`PacketQuoteTable,`PacketQuoteSemantics].contains modules[index]! then
        let info ← getConstInfo name
        let declaration ← match info with
          | .thmInfo declaration =>
            theorems := theorems.push name
            pure (Declaration.thmDecl {declaration with name := ← mkFreshUserName `__packetBody})
          | .defnInfo declaration =>
            definitions := definitions.push name
            pure (Declaration.defnDecl {declaration with name := ← mkFreshUserName `__packetData})
          | _ => continue
        let options := (← getOptions).setBool `debug.skipKernelTC false
        let environment := Kernel.enableDiag (Kernel.resetDiag (← getEnv)) true
        let env ← ofExceptKernelException <| environment.addDeclCore
          (Core.getMaxHeartbeats options).toUSize (maxRecDepth.get options).toUSize
          declaration none (doCheck := true)
        let diagnostics := Kernel.getDiagnostics env
        unless diagnostics.enabled do throwError "packet quotation diagnostics disabled"
        if let some (producer,count) := Hex.RCF.SelectedRootTests.KernelCheck.violation? diagnostics
            #[``ScalarPacket.build,``ScalarPacket.buildLiteral] then
          throwError "packet quotation executed native producer {producer}: {count}; body={name}"
        if !diagnostics.unfoldCounter.isEmpty then reductions := reductions + 1
        setEnv (Kernel.enableDiag env false)
  unless reductions > 0 do throwError "packet quotation audit did not observe any reductions"
  unless theorems.size == 70 && definitions.size == 14 do
    throwError "packet quotation body coverage changed: {theorems.size}/{definitions.size}"
  logInfo m!"packet quotation: {theorems.size} theorem bodies, {definitions.size} data definitions; complete standard-three axiom closure; no native producer reductions"
  logInfo m!"packet theorem inventory: {theorems.qsort Name.lt}"
  logInfo m!"packet data inventory: {definitions.qsort Name.lt}"
end Hex.RCF.PacketQuoteTests
